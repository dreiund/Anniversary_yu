- [ ] fetchCouple 确定性:
```swift
    /// P6-B1:多 couple 并存时确定性选择——共享 store 里的优先(配对完成语义强),
    /// 同 store 按 createdAt 最早;彻底告别无排序 fetch 的随机首个(P2 双 couple 事故防线)
    func fetchCouple() throws -> CDCouple? {
        let request = CDCouple.fetchRequest()
        let all = (try context.fetch(request) as? [CDCouple]) ?? []
        guard all.count > 1 else { return all.first }
        let shared = PersistenceController.shared.sharedStore
        func inShared(_ c: CDCouple) -> Bool {
            c.objectID.persistentStore === shared
        }
        return all.sorted { a, b in
            if inShared(a) != inShared(b) { return inShared(a) }
            return (a.createdAt ?? .distantFuture) < (b.createdAt ?? .distantFuture)
        }.first
    }
```
(CDCouple 若无 createdAt 字段,tiebreak 换 `id?.uuidString` 字典序——先查 ModelSchema,禁止为此加字段。)
